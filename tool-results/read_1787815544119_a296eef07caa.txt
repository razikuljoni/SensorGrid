   168→function paletteEntryFor(type: string, kind: string): NodePaletteEntry {
   169→  return (
   170→    NODE_PALETTE.find((p) => p.type === type && p.kind === kind) ??
   171→    NODE_PALETTE.find((p) => p.type === type) ??
   172→    NODE_PALETTE[0]
   173→  )
   174→}

   176→function summarizeConfig(node: AutomationNode): string {
   177→  const c = node.data.config ?? {}
   178→  const t = node.type
   179→  switch (t) {
   180→    case 'trigger': {
   181→      if (node.data.kind === 'schedule') return `cron ${String(c.cron ?? '—')}`
   182→      if (node.data.kind === 'manual') return 'manual trigger'
   183→      if (node.data.kind === 'telemetry_received') return `${c.sensorKey ?? 'any'} on ${c.deviceId ? String(c.deviceId).slice(0, 8) : 'any'}`
   184→      return c.deviceId ? `device ${String(c.deviceId).slice(0, 8)}` : 'any device'
   185→    }
   186→    case 'condition':
   187→      return `${c.sensorKey ?? 'value'} ${node.data.kind} ${c.threshold ?? '—'}`
   188→    case 'logic':
   189→      return node.data.kind.toUpperCase()
   190→    case 'delay':
   191→      return `${c.seconds ?? 0}s`
   192→    case 'action': {
   193→      if (node.data.kind === 'send_command') return `→ ${c.deviceId ? String(c.deviceId).slice(0, 8) : 'device'}`
   194→      return `twin ${c.deviceId ? String(c.deviceId).slice(0, 8) : 'device'}`
   195→    }
   196→    case 'notification':
   197→      return c.title ? String(c.title) : c.category ? String(c.category) : 'notify'
   198→    default:
   199→      return ''
   200→  }
   201→}

   218→  const Icon = entry.icon
   219→  const accent = NODE_ACCENT_CLASS[entry.accent]
   220→  const summary = summarizeConfig({ id, type: d.type, position: { x: 0, y: 0 }, data: { label: d.label, kind: d.kind, config: d.config } })
   221→
   222→  return (
   223→    <div
   224→      className={cn(
   225→        'group relative w-56 rounded-xl border bg-card text-card-foreground shadow-md transition-all',

   322→    return (
   323→      <div className="flex h-full flex-col items-center justify-center gap-2 p-6 text-center">

   326→        </div>
   327→        <p className="text-sm font-medium text-text-secondary">No node selected</p>

   331→  }
   332→
   333→  const d = node.data as PulseNodeData
   334→  const entry = paletteEntryFor(d.type, d.kind)
   335→  const accent = NODE_ACCENT_CLASS[entry.accent]
   336→  const Icon = entry.icon

   338→
   339→  const setConfig = (key: string, value: unknown) => {
   340→    onChange(node.id, { config: { ...cfg, [key]: value } })
   341→  }
   342→
   343→  return (
   344→    <div className="flex h-full flex-col">
   345→      <div className="flex items-center gap-2 px-4 py-3 border-b border-border">
   346→        <span className={cn('flex size-8 items-center justify-center rounded-lg', accent.chip)}>
   347→          <Icon className="size-4" />

   350→          <p className="text-xs uppercase tracking-wide text-text-muted">{entry.type}</p>
   351→          <p className="text-sm font-semibold">{entry.label}</p>
   352→        </div>
   353→      </div>
   354→      <ScrollArea className="flex-1">
   355→        <div className="flex flex-col gap-4 p-4">
   356→          <div className="flex flex-col gap-1.5">
   357→            <Label htmlFor="insp-label" className="text-xs text-text-muted">Label</Label>
   358→            <Input
   359→              id="insp-label"
   360→              value={d.label}
   361→              onChange={(e) => onChange(node.id, { label: e.target.value })}
   362→              className="h-8 text-sm"
   363→            />
   364→          </div>
   365→
   366→          <div className="flex flex-col gap-1.5">
   367→            <Label className="text-xs text-text-muted">Sub-type</Label>
   368→            <div className="font-mono text-xs bg-muted px-2 py-1.5 rounded-md">{d.kind}</div>
   369→          </div>
   370→
   371→          <Separator />
   372→
   373→          {/* Per-kind config */}
   374→          {d.type === 'trigger' && d.kind === 'telemetry_received' && (
   375→            <>
   376→              <ConfigInput label="Device ID" value={cfg.deviceId} onChange={(v) => setConfig('deviceId', v)} placeholder="any" />
   377→              <ConfigInput label="Sensor Key" value={cfg.sensorKey} onChange={(v) => setConfig('sensorKey', v)} placeholder="temperature" />

   381→            <ConfigInput label="Device ID" value={cfg.deviceId} onChange={(v) => setConfig('deviceId', v)} placeholder="any" />
   382→          )}
   383→          {d.type === 'trigger' && d.kind === 'schedule' && (
   384→            <ConfigInput label="Cron Expression" value={cfg.cron} onChange={(v) => setConfig('cron', v)} placeholder="0 * * * *" mono />

   395→            <ConfigNumber label="Seconds" value={cfg.seconds as number} onChange={(v) => setConfig('seconds', v)} />

   412→            </>

   444→    </div>

   446→}

   560→  automation: AutomationDTO

   564→function RuleCanvasInner({ automation, onSaved }: RuleCanvasProps) {
   565→  const qc = useQueryClient()
   566→  const [nodes, setNodes, onNodesChange] = useNodesState(automation.nodes.map(toFlowNode))
   567→  const [edges, setEdges, onEdgesChange] = useEdgesState(automation.edges.map(toFlowEdge))
   568→  const [selectedNodeId, setSelectedNodeId] = React.useState<string | null>(null)
   569→  const [saving, setSaving] = React.useState(false)
   570→  const [dirty, setDirty] = React.useState(false)
   571→  const rf = useReactFlow()
   572→
   573→  // Re-sync when automation changes (e.g. switching selected automation)
   574→  React.useEffect(() => {
   575→    setNodes(automation.nodes.map(toFlowNode))
   576→    setEdges(automation.edges.map(toFlowEdge))
   577→    setSelectedNodeId(null)
   578→    setDirty(false)
   579→  }, [automation.id, automation.version, setNodes, setEdges])
   580→
   581→  const onConnect = React.useCallback(
   582→    (c: Connection) => {
   583→      setEdges((eds) => addEdge({ ...c, animated: true, id: `e_${makeNodeId()}` }, eds))
   584→      setDirty(true)
   585→    },
   586→    [setEdges]
   587→  )
   588→
   589→  const addNode = (entry: NodePaletteEntry) => {
   590→    const center = rf.screenToFlowPosition({ x: window.innerWidth / 2, y: window.innerHeight / 2 })
   591→    // Fallback to a pseudo-random offset if screenToFlowPosition is unavailable
   592→    const pos = center && Number.isFinite(center.x) && Number.isFinite(center.y)
   593→      ? center
   594→      : { x: 80 + Math.random() * 200, y: 80 + Math.random() * 160 }
   595→    const id = makeNodeId()
   596→    const newNode: Node = {
   597→      id,
   598→      type: entry.type,
   599→      position: pos,
   600→      data: { label: entry.label, kind: entry.kind, config: { ...entry.defaultConfig }, type: entry.type },
   601→    }
   602→    setNodes((nds) => [...nds, newNode])
   603→    setSelectedNodeId(id)
   604→    setDirty(true)
   605→    toast.success(`Added "${entry.label}" node`)
   606→  }
   607→
   608→  const updateNodeData = React.useCallback((id: string, patch: Partial<PulseNodeData>) => {
   609→    setNodes((nds) =>
   610→      nds.map((n) => {
   611→        if (n.id !== id) return n
   612→        const merged = { ...n.data, ...patch, config: patch.config ?? { ...(n.data as PulseNodeData).config, ...(patch.config ?? {}) } }
   613→        return { ...n, data: merged }
   614→      })
   615→    )
   616→    setDirty(true)
   617→  }, [setNodes])
   618→
   619→  const deleteNode = React.useCallback((id: string) => {
   620→    setNodes((nds) => nds.filter((n) => n.id !== id))
   621→    setEdges((eds) => eds.filter((e) => e.source !== id && e.target !== id))
   622→    setSelectedNodeId(null)
   623→    setDirty(true)
   624→  }, [setNodes, setEdges])
   625→
   626→  const save = async () => {
   627→    setSaving(true)
   628→    try {
   629→      const payload = {
   630→        nodes: nodes.map(toAutomationNode),
   631→        edges: edges.map(toAutomationEdge),
   632→      }
   633→      const r = await fetch(`/api/automations/${automation.id}`, {
   634→        method: 'PATCH',
   635→        headers: { 'Content-Type': 'application/json' },
   636→        body: JSON.stringify(payload),
   637→      })
   638→      if (!r.ok) throw new Error('Failed to save')
   639→      toast.success('Automation saved', { description: `Version ${automation.version + 1}` })
   640→      setDirty(false)
   641→      qc.invalidateQueries({ queryKey: qk.automations })
   642→      qc.invalidateQueries({ queryKey: qk.automation(automation.id) })
   643→      onSaved()
   644→    } catch (err) {
   645→      toast.error('Could not save automation', { description: err instanceof Error ? err.message : 'Unknown error' })
   646→    } finally {
   647→      setSaving(false)
   648→    }
   649→  }
   650→
   651→  const selectedNode = React.useMemo(
   652→    () => nodes.find((n) => n.id === selectedNodeId) ?? null,
   653→    [nodes, selectedNodeId]
   654→  )
   655→
   656→  return (
   657→    <div className="grid grid-cols-1 lg:grid-cols-[200px_1fr_280px] gap-0 h-full min-h-[600px] border-t border-border">
   658→      {/* Palette */}
   659→      <div className="hidden lg:block border-r border-border bg-background/40">
   660→        <PaletteSidebar onAdd={addNode} />
   661→      </div>
   662→
   663→      {/* Canvas */}
   664→      <div className="relative">
   665→        <div className="absolute left-3 top-3 z-10 flex items-center gap-2">
   666→          <Badge variant="outline" className="bg-background/80 backdrop-blur">
   667→            <Workflow className="size-3" />
   668→            {automation.name}
   669→          </Badge>
   670→          {dirty && (
   671→            <Badge className="bg-warning/10 text-warning border-warning/30 border">
   672→              <span className="size-1.5 rounded-full bg-warning" /> Unsaved
   673→            </Badge>
   674→          )}
   675→        </div>
   676→        <div className="absolute right-3 top-3 z-10 flex items-center gap-2">
   677→          {/* Mobile palette trigger */}
   678→          <div className="lg:hidden">
   679→            <NodeQuickAddMobile onAdd={addNode} />
   680→          </div>
   681→          <Button size="sm" onClick={save} disabled={saving || !dirty}>
   682→            <Save className="size-3.5" />
   683→            {saving ? 'Saving...' : 'Save'}
   684→          </Button>
   685→        </div>
   686→        <ReactFlow
   687→          nodes={nodes}
   688→          edges={edges}
   689→          onNodesChange={onNodesChange}
   690→          onEdgesChange={onEdgesChange}
   691→          onConnect={onConnect}
   692→          nodeTypes={nodeTypes}
   693→          onNodeClick={(_, n) => setSelectedNodeId(n.id)}
   694→          onPaneClick={() => setSelectedNodeId(null)}
   695→          fitView
   696→          fitViewOptions={{ padding: 0.2 }}
   697→          proOptions={{ hideAttribution: true }}
   698→          className="bg-muted/30"
   699→        >
   700→          <Background gap={16} size={1} className="opacity-40" />
   701→          <Controls className="!bg-background !border-border !shadow-md" />
   702→          <MiniMap
   703→            className="!bg-background !border !border-border rounded-md"
   704→            nodeColor={(n) => {
   705→              const d = n.data as PulseNodeData
   706→              const entry = paletteEntryFor(d.type, d.kind)
   707→              const accent = NODE_ACCENT_CLASS[entry.accent]
   708→              return accent.dot.replace('bg-', '')
   709→            }}
   710→            pannable
   711→            zoomable
   712→          />
   713→        </ReactFlow>
   714→      </div>
   715→
   716→      {/* Inspector */}
   717→      <div className="hidden lg:block border-l border-border bg-background/40">
   718→        <InspectorPanel node={selectedNode} onChange={updateNodeData} onDelete={deleteNode} />
   719→      </div>
   720→
   721→      {/* Mobile inspector */}
   722→      {selectedNode && (
   723→        <div className="lg:hidden border-t border-border bg-background max-h-[40vh]">
   724→          <InspectorPanel node={selectedNode} onChange={updateNodeData} onDelete={deleteNode} />
   725→        </div>
   726→      )}
   727→    </div>
   728→  )
   729→}
